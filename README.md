		$ npm install express -s
iniciar git
	$ git init
	$ git branch -M main
	$ git remote add origin https://github.com/memoriasrp/ServerSide_RedBicis.git
	$ git push -u origin main

	$ npm install express-generator -g

	$ express red-bicicletas --view=pug
